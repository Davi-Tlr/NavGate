import { useState, useCallback } from 'react';
import { Aerodromo } from '../../aerodromos/types';
import { Waypoint, Rota } from '../types';
import { calcularRota, rotaParaGeoJSON } from '../services/rotaService';

export function useRota() {
    const [waypoints, setWaypoints] = useState<Waypoint[]>([]);

    const adicionarWaypoint = useCallback((aerodromo: Aerodromo) => {
        setWaypoints(prev => {
            if (prev.length > 0 && prev[prev.length - 1]?.aerodromo?.icao === aerodromo.icao) {
                return prev;
            }
            return [...prev, {
                id: `${aerodromo.icao}-${Date.now()}`,
                aerodromo,
                label: aerodromo.icao,
            }];
        });
    }, []);

    const removerWaypoint = useCallback((id: string) => {
        setWaypoints(prev => prev.filter(w => w.id !== id));
    }, []);

    const limparRota = useCallback(() => setWaypoints([]), []);

    const rota: Rota | null = waypoints.length >= 2 ? calcularRota(waypoints) : null;
    const linhaGeoJSON = rotaParaGeoJSON(waypoints);

    return { waypoints, rota, linhaGeoJSON, adicionarWaypoint, removerWaypoint, limparRota };
}
